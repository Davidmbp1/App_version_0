rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    // Permitir acceso público SOLO a la carpeta videos_onb_instr
    match /videos_onb_instr/{file} {
      allow read: if true; // Permitir lectura pública sin autenticación
    }
    // Permitir acceso público a la carpeta "Revision de callos" para las imágenes
    // SOLUCIÓN: Usar una regla que capture paths que empiecen con "Revision" y contengan "callos"
    // Esto funciona porque Firebase Storage almacena los paths con espacios como están
    match /{path=**} {
      // Permitir lectura si el path contiene "Revision de callos" (con espacios)
      allow read: if path.matches('.*Revision de callos.*');
      // Requiere autenticación para escribir
      allow write: if request.auth != null;
    }
    // Regla más específica para el resto (requiere autenticación)
    match /{allPaths=**} {
      allow read, write: if request.auth != null;
    }
  }
}
